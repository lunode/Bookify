# Application Layer and CQRS

## Creating the Application project

```sh
cd Bookify
dotnet new classlib -o src/Bookify.Application
dotnet sln add src/Bookify.Application
```

## Configuring dependency injection and installing MediatR

```sh
cd src/Bookify.Application
dotnet add package MediatR -v 12.5.0
dotnet add reference ../Bookify.Domain
```

- DependencyInjection.cs

```cs
using Bookify.Domain.Bookings;
using Microsoft.Extensions.DependencyInjection;

namespace Bookify.Application;


public static class DependencyInjection
{
    public static IServiceCollection AddApplication(this IServiceCollection services)
    {
        services.AddMediatR(config =>
        {
            config.RegisterServicesFromAssembly(typeof(DependencyInjection).Assembly);
        });

        services.AddTransient<PricingService>();

        return services;
    }
}
```

## Adding the required abstractions for CQRS

- Abstractions
  - Messaging
    - IQuery.cs
    - IQueryHandler.cs
    - ICommand.cs
    - ICommandHandler.cs
    

```cs
using Bookify.Domain.Abstractions;
using MediatR;

namespace Bookify.Application.Abstractions.Messaging;

public interface IQuery<TResult> : IRequest<Result<TResult>>
{
}
```

```cs
using Bookify.Domain.Abstractions;
using MediatR;

namespace Bookify.Application.Abstractions.Messaging;

public interface IQueryHandler<TQuery, TResponse> : IRequestHandler<TQuery, Result<TResponse>>
    where TQuery : IQuery<TResponse>
{
}
```

```cs
using Bookify.Domain.Abstractions;
using MediatR;

namespace Bookify.Application.Abstractions.Messaging;

public interface ICommand : IRequest<Result>, IBaseCommand
{
}

public interface ICommand<TResponse> : IRequest<Result<TResponse>>, IBaseCommand
{
}

public interface IBaseCommand { }
```

```cs
using Bookify.Domain.Abstractions;
using MediatR;

namespace Bookify.Application.Abstractions.Messaging;

public interface ICommandHandler<TCommand> : IRequestHandler<TCommand, Result>
    where TCommand : ICommand
{
}

public interface ICommandHandler<TCommand, TResponse> : IRequestHandler<TCommand, Result<TResponse>>
    where TCommand : ICommand<TResponse>
{
}
```
